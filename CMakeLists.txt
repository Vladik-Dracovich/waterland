cmake_minimum_required(VERSION 3.16)
project(WaterRenderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenGL REQUIRED)
find_package(glfw3 REQUIRED)
find_package(glm REQUIRED)
add_executable(water main.cpp)
target_link_libraries(water PRIVATE OpenGL::GL glfw glm::glm)
configure_file(shaders/water.vert shaders/water.vert COPYONLY)
configure_file(shaders/water.frag shaders/water.frag COPYONLY)
